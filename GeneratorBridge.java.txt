        /** All Excel data rows as JSON array for skip logic */
        @JavascriptInterface
        public String listExcelRows() {
            try {
                if (cachedExcelPath == null)
                    return "[]";
                if (!Python.isStarted())
                    Python.start(new AndroidPlatform(MainActivity.this));
                Python py = Python.getInstance();
                PyObject gen = py.getModule("generator");
                PyObject result = gen.callAttr("list_data_rows", cachedExcelPath);
                return result.toString();
            } catch (Exception e) {
                return "{\"error\":\"" + e.getMessage().replace("\"", "'") + "\"}";
            }
        }

        /**
         * Generate one docx for Excel sheet row number.
         * Returns OK:/abs/path.docx or ERROR:message
         */
        @JavascriptInterface
        public String generateOneRow(String excelRowNum) {
            try {
                if (cachedExcelPath == null || cachedWordPath == null)
                    return "ERROR:Files not loaded";
                if (!Python.isStarted())
                    Python.start(new AndroidPlatform(MainActivity.this));

                File out = new File(getCacheDir(),
                        "row_" + excelRowNum + "_" + System.currentTimeMillis() + ".docx");

                Python py = Python.getInstance();
                PyObject gen = py.getModule("generator");
                PyObject result = gen.callAttr(
                        "generate_one_row",
                        cachedExcelPath,
                        cachedWordPath,
                        excelRowNum,
                        out.getAbsolutePath());

                boolean ok = result.asList().get(0).toBoolean();
                String msg = result.asList().get(1).toString();
                if (!ok) return "ERROR:" + msg;
                return "OK:" + out.getAbsolutePath();
            } catch (Exception e) {
                return "ERROR:" + e.getMessage();
            }
        }

        /**
         * Merge list of docx paths, publish to Downloads.
         * pathsJson: ["path1","path2"]
         * Returns OK:fileName|publishedPath or ERROR:...
         */
        @JavascriptInterface
        public String mergeAndPublishBatch(String pathsJson) {
            try {
                if (!Python.isStarted())
                    Python.start(new AndroidPlatform(MainActivity.this));

                String fileName = "DocsPlus_Batch_" + System.currentTimeMillis() + ".docx";
                File merged = new File(getCacheDir(), fileName);

                Python py = Python.getInstance();
                PyObject gen = py.getModule("generator");
                PyObject result = gen.callAttr(
                        "merge_docx_list",
                        pathsJson,
                        merged.getAbsolutePath());

                boolean ok = result.asList().get(0).toBoolean();
                String msg = result.asList().get(1).toString();
                if (!ok) return "ERROR:" + msg;

                String published = publishToDownloads(merged, fileName);

                // cleanup temps listed in pathsJson + merged cache copy
                try {
                    // delete individual row temps
                    String trimmed = pathsJson.trim();
                    if (trimmed.startsWith("[")) {
                        // best-effort cleanup via Python list already done; delete files
                    }
                } catch (Exception ignored) {}
                try { merged.delete(); } catch (Exception ignored) {}

                // Also try delete each path from JSON manually
                try {
                    // simple split cleanup without full JSON parser
                    String s = pathsJson.replace("[", "").replace("]", "").replace("\"", "");
                    for (String p : s.split(",")) {
                        p = p.trim();
                        if (p.length() > 0) {
                            File f = new File(p);
                            if (f.exists()) f.delete();
                        }
                    }
                } catch (Exception ignored) {}

                return "OK:" + fileName + "|" + published;
            } catch (Exception e) {
                return "ERROR:" + e.getMessage();
            }
        }
